cmake_minimum_required(VERSION 3.18)
project(maloni_physics LANGUAGES CXX)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_path(BULLET_COLLISION_INCLUDE btBulletCollisionCommon.h PATH_SUFFIXES bullet REQUIRED)
find_library(BULLET_COLLISION_LIBRARY BulletCollision REQUIRED)
find_library(BULLET_MATH_LIBRARY LinearMath REQUIRED)
add_library(maloni_physics SHARED engine.cpp)
target_include_directories(maloni_physics SYSTEM PRIVATE ${BULLET_COLLISION_INCLUDE})
target_link_libraries(maloni_physics PRIVATE Eigen3::Eigen ${BULLET_COLLISION_LIBRARY} ${BULLET_MATH_LIBRARY})
target_compile_features(maloni_physics PRIVATE cxx_std_17)
target_compile_options(maloni_physics PRIVATE -Wall -Wextra -Wpedantic)
